/* Bộ giải chạy trong Web Worker. Nếu WebAssembly lỗi (panic, hết bộ nhớ, không tải được) thì mọi lời gọi đang chờ
   được báo lỗi ngay thay vì treo mãi, và worker tự khởi động lại (tối đa 3 lần trong 30 giây để tránh lặp vô hạn). */
let W = null,
  wid = 0,
  pend = {},
  crashes = [];
const failAll = m => {
  const ps = Object.values(pend);
  pend = {};
  ps.forEach(p => p.rej(new Error(m)));
};
function startWorker() {
  W = new Worker('worker.js', { type: 'module' });
  W.onmessage = e => {
    const d = e.data;
    if (d.ready) {
      say('Sẵn sàng.');
      return;
    }
    if (d.fatal) {
      W.terminate();
      W = null;
      failAll(d.fatal);
      say(d.fatal + ' Hãy tải lại trang.');
      return;
    }
    const p = pend[d.id];
    delete pend[d.id];
    if (p) {
      const j = JSON.parse(d.out);
      j.ok ? p.res(j.data) : p.rej(new Error(j.error));
    }
    if (d.crashed) restartWorker('Bộ giải gặp lỗi nội bộ và đã được khởi động lại.');
  };
  W.onerror = e => {
    if (e && e.preventDefault) e.preventDefault();
    restartWorker('Lỗi WebAssembly: ' + ((e && e.message) || 'không tải được') + '.');
  };
}
function restartWorker(msg) {
  const now = Date.now();
  crashes = crashes.filter(t => now - t < 30000);
  crashes.push(now);
  if (W) W.terminate();
  W = null;
  failAll(msg);
  if (crashes.length > 3) {
    say(msg + ' Bộ giải lỗi liên tục, hãy tải lại trang.');
    return;
  }
  say(msg + ' Đang khởi tạo lại…');
  startWorker();
}
startWorker();
const post = (msg, tr) =>
  new Promise((res, rej) => {
    if (!W) return rej(new Error('Bộ giải chưa sẵn sàng, hãy tải lại trang.'));
    const id = ++wid;
    pend[id] = { res, rej };
    W.postMessage(Object.assign({ id }, msg), tr || []);
  });
const api = (route, body) => post({ route, body });
const frameApi = (w, h, buf, s) => post({ frame: { w, h, s }, buf }, [buf]);

  startWorker();
  say('Đã huỷ phép tính, đang khởi tạo lại bộ giải…');
}

  const { data, error } = await (await sb()).rpc(fn, args || {});
