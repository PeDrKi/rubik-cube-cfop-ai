/* Gọi server: báo lỗi tiếng Việt dễ hiểu cả khi mất mạng hoặc server/proxy trả trang lỗi không phải JSON (502, 413…). */
async function req(url, opt) {
  let r,
    j = null;
  try {
    r = await fetch(url, opt);
  } catch (e) {
    throw new Error('Mất kết nối tới máy chủ, kiểm tra mạng rồi thử lại.');
  }
  try {
    j = await r.json();
  } catch (e) {}
  if (r.ok && j) return j;
  throw new Error(
    (j && j.error) ||
      (r.status === 429
        ? 'Bạn thao tác quá nhanh, hãy thử lại sau ít phút.'
        : r.status === 413
          ? 'Dữ liệu gửi lên quá lớn.'
          : r.status >= 500
            ? `Máy chủ đang gặp sự cố (mã ${r.status}), hãy thử lại sau.`
            : `Yêu cầu không hợp lệ (mã ${r.status}).`)
  );
}
